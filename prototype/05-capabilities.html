// TECH STACK · 完全开源，自主可控
$ cat platform-stack.yml

# 智能体层
agent_framework:   DeepAgents / LangGraph   # MIT License
reasoning_model:   deepseek-v4-pro          # 国产前沿模型

# 执行层
sandbox_runtime:   Docker + gVisor (runsc)  # Google 开源
event_transport:   SSE (Server-Sent Events) # W3C 标准
task_queue:        Redis Streams            # 持久化事件流

# 数据层
checkpoint_store:  PostgreSQL               # 状态持久化
artifact_store:    MinIO (S3-compatible)    # 院内对象存储

# 接入层
api_layer:         FastAPI (Python)
frontend:          React + Vite + TypeScript
deployment:        Docker Compose · 单机内网部署
全部组件均可在学院服务器独立运行，无需连接任何外部商业服务
